option(CIDER_BUILD_HOST_PROBE "Build an isolated host OpenGL rendering probe" ON)
if(CIDER_BUILD_HOST_PROBE)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(CIDER_PROBE_EGL QUIET IMPORTED_TARGET egl)
        pkg_check_modules(CIDER_PROBE_GL QUIET IMPORTED_TARGET gl)
    endif()
    if(TARGET PkgConfig::CIDER_PROBE_EGL AND TARGET PkgConfig::CIDER_PROBE_GL)
        add_executable(cider-gl-probe graphics-probe.c)
        target_compile_options(cider-gl-probe PRIVATE -Wall -Wextra -pthread)
        target_link_libraries(cider-gl-probe PRIVATE PkgConfig::CIDER_PROBE_EGL PkgConfig::CIDER_PROBE_GL pthread)
        install(TARGETS cider-gl-probe DESTINATION libexec/ciderdarling)
    else()
        message(STATUS "Cider host graphics probe unavailable: install EGL/OpenGL development packages")
    endif()
endif()
